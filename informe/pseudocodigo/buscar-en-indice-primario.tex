funcion buscarEnIndicePrimario(carne):
    posicion = busquedaBinaria(zonaPrimaria, carne)
    si posicion != NO_ENCONTRADO:
        retornar posicion

    para i desde 0 hasta longitud(zonaAltasNuevas) - 1:
        si zonaAltasNuevas[i].carne == carne:
            retornar zonaAltasNuevas[i].posicion

    retornar NO_ENCONTRADO
\end{lstlisting}
